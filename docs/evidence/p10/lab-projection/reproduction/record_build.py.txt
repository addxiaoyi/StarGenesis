from pathlib import Path
import hashlib, json, os, sys
import xml.etree.ElementTree as ET
work = Path(__file__).resolve().parent
root = work.parents[1]
label = sys.argv[1]
reports = sorted(root.glob('*/target/surefire-reports/TEST-*.xml'))
counts = {key: sum(int(ET.parse(path).getroot().attrib[key]) for path in reports)
          for key in ('tests','failures','errors','skipped')}
assert counts == {'tests': 584, 'failures':0, 'errors':0, 'skipped':0}, counts
log = work/f'build-{label}.log'
assert '[INFO] BUILD SUCCESS' in log.read_text(encoding='utf-8', errors='replace')
artifacts = [root/'stargenesis-paper/target/stargenesis-paper-4.0.0-SNAPSHOT.jar',
             root/'stargenesis-helstera/target/stargenesis-helstera-4.0.0-SNAPSHOT.jar', root/'target/sbom.json', log]
summary = {'tests':counts, 'suites':len(reports), 'javaHome':os.environ['JAVA_HOME'], 'mavenOpts':os.environ['MAVEN_OPTS'],
           'command':'mvnw.cmd -B -ntp -Phelstera-bridge -DforkCount=0 clean verify',
           'sha256':{str(path.relative_to(root)):hashlib.sha256(path.read_bytes()).hexdigest() for path in artifacts}}
(work/f'build-{label}.json').write_text(json.dumps(summary,indent=2)+'\n',encoding='utf-8')
print(json.dumps(summary))
