package dev.stargenesis.paper.arena;

import java.io.IOException;
import java.util.HashMap;
import java.util.Map;
import java.util.UUID;

/** In-memory Arena invitations; team membership is committed by the match service. */
final class ArenaInvitations implements AutoCloseable {
    static final int MAX_INVITATIONS = 64;
    private final Map<UUID, Invitation> pending = new HashMap<>();

    synchronized void invite(UUID inviter, UUID invited, String team) {
        boolean inviterHasPending = pending.values().stream().anyMatch(existing -> existing.inviter().equals(inviter));
        if (pending.containsKey(invited) || inviterHasPending)
            throw new IllegalStateException("你或受邀玩家已有待处理的 Arena 邀请。");
        pending.put(invited, new Invitation(inviter, team));
    }

    Invitation accept(UUID invited, TeamJoiner joiner) throws IOException {
        Invitation invitation;
        synchronized (this) { invitation = pending.remove(invited); }
        if (invitation == null) throw new IllegalStateException("没有可接受的有效 Arena 邀请。");
        try { joiner.join(invitation.inviter(), invited, invitation.team()); }
        catch (IOException failure) {
            synchronized (this) { pending.putIfAbsent(invited, invitation); }
            throw failure;
        }
        return invitation;
    }

    synchronized void cancel(UUID player) {
        pending.remove(player);
        pending.entrySet().removeIf(entry -> entry.getValue().inviter().equals(player));
    }

    @Override public synchronized void close() { pending.clear(); }

    record Invitation(UUID inviter, String team) { }
    @FunctionalInterface interface TeamJoiner { void join(UUID inviter, UUID invited, String team) throws IOException; }
}
